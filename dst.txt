
 1. COMMON STARTING COMMANDS (Run these first on EVERY device)

enable                                        ! Enter privileged EXEC mode
configure terminal                            ! Enter global configuration mode


 2. ROUTER SPECIFIC: RTA

hostname RTA                                  ! Set device name
interface g0/0                                ! Enter GigabitEthernet 0/0 interface
 ip address 172.16.1.1 255.255.255.0          ! Assign IP address and subnet mask
 no shutdown                                  ! Turn on the interface
 exit                                         ! Return to global config
service password-encryption                   ! Encrypt all plaintext passwords
security passwords min-length 10              ! Set minimum password length to 10
enable secret itexamanswers                   ! Set strong encrypted enable password
no ip domain-lookup                           ! Disable accidental DNS translation delays
ip domain-name CCNA.com                       ! Set domain name for SSH keys
username jony secret itexamanswers.net        ! Create local user for remote SSH access
crypto key generate rsa                       ! Generate encryption keys
1024                                          ! Specify 1024-bit key size (responds to prompt)
login block-for 180 attempts 4 within 120     ! Block brute-force logins for 3 mins
line vty 0 4                                  ! Configure remote access lines
 transport input ssh                          ! Only allow SSH connections (block Telnet)
 login local                                  ! Require the local username and password
 exec-timeout 6                               ! Disconnect idle users after 6 minutes
 exit                                         ! Return to global config


 3. SWITCH SPECIFIC: SW1

hostname SW1                                  ! Set device name
interface vlan 1                              ! Enter management VLAN interface
 ip address 172.16.1.2 255.255.255.0          ! Assign management IP address
 no shutdown                                  ! Turn on the VLAN interface
 exit                                         ! Return to global config
ip default-gateway 172.16.1.1                 ! Point to the router to reach other networks
interface range f0/2 - 24, g0/2               ! Select all unused ports at once
 shutdown                                     ! Disable unused ports for security
 exit                                         ! Return to global config
service password-encryption                   ! Encrypt all plaintext passwords
enable secret class                           ! Set strong encrypted enable password
no ip domain-lookup                           ! Disable accidental DNS translation delays
ip domain-name CCNA.com                       ! Set domain name for SSH keys
crypto key generate rsa                       ! Generate encryption keys
1024                                          ! Specify 1024-bit key size (responds to prompt)
username admin_switch secret p@ssword         ! Create local user for remote SSH access
line vty 0 15                                 ! Configure all remote access lines
 transport input ssh                          ! Only allow SSH connections
 login local                                  ! Require the local username and password
 exec-timeout 6                               ! Disconnect idle users after 6 minutes
 exit                                         ! Return to global config


 4. FIREWALL SPECIFIC: NETSEC-ASA

hostname NETSEC-ASA                           ! Set device name
domain-name netsec.com                        ! Set domain name for SSH keys
enable password ciscoenpa55                   ! Set enable password
clock set 10:37:27 April 13 2021              ! Set the system clock manually
interface g1/2                                ! Enter GigabitEthernet 1/2 configuration
 nameif INSIDE                                ! Name the interface INSIDE
 ip address 192.168.1.1 255.255.255.0         ! Assign internal IP address and mask
 security-level 100                           ! Set security level to highest trust (100)
 no shutdown                                  ! Turn on the interface
interface g1/1                                ! Enter GigabitEthernet 1/1 configuration
 nameif OUTSIDE                               ! Name the interface OUTSIDE
 ip address 209.165.200.226 255.255.255.248   ! Assign external IP address and mask
 security-level 0                             ! Set security level to lowest trust (0)
 no shutdown                                  ! Turn on the interface
route OUTSIDE 0.0.0.0 0.0.0.0 209.165.200.225 ! Create default static route out to ISP
object network INSIDE-NET                     ! Create a network object for the LAN
 subnet 192.168.1.0 255.255.255.0             ! Define the LAN subnet
 nat (INSIDE,OUTSIDE) dynamic interface       ! Configure PAT (hide internal IPs behind OUTSIDE IP)
dhcpd address 192.168.1.5-192.168.1.36 INSIDE ! Set DHCP pool range for internal clients
dhcpd dns 209.165.201.2 interface INSIDE      ! Set DNS server given to DHCP clients
dhcpd enable INSIDE                           ! Turn on DHCP server on INSIDE interface
username admin password adminpa55             ! Create local user for remote administration
aaa authentication ssh console LOCAL          ! Require local database for SSH logins
crypto key generate rsa modulus 1024          ! Generate 1024-bit RSA keys for SSH
no                                            ! (Responds 'no' if asked to replace existing keys)
ssh 192.168.1.0 255.255.255.0 INSIDE          ! Allow SSH from any inside network PC
ssh 172.16.3.3 255.255.255.255 OUTSIDE        ! Restrict outside SSH to this specific IP only
ssh timeout 10                                ! Disconnect idle SSH sessions after 10 minutes
interface g1/3                                ! Enter GigabitEthernet 1/3 configuration
 ip address 192.168.2.1 255.255.255.0         ! Assign DMZ IP address
 nameif DMZ                                   ! Name the interface DMZ
 security-level 70                            ! Set security level to medium trust (70)
 no shutdown                                  ! Turn on the interface
object network DMZ-SERVER                     ! Create network object for the web server
 host 192.168.2.3                             ! Specify the internal IP of the DMZ server
 nat (DMZ,OUTSIDE) static 209.165.200.227     ! Map the server's private IP to a public IP (Static NAT)
access-list OUTSIDE-DMZ permit icmp any host 192.168.2.3      ! Allow pings from internet to DMZ server
access-list OUTSIDE-DMZ permit tcp any host 192.168.2.3 eq 80 ! Allow HTTP from internet to DMZ server
access-group OUTSIDE-DMZ in interface OUTSIDE                 ! Apply access list to inbound traffic on OUTSIDE
exit                                                          ! Return to global config

! 6. ROUTER 1 SPECIFIC (R1) - PART B: VPN and ACL Config (Run after reboot)

enable                                        ! Re-enter privileged EXEC mode
configure terminal                            ! Re-enter global config mode
ip access-list standard File_Server_Restrictions  ! Create the Standard ACL you requested
 permit host 192.168.20.4                     ! Allow this specific IP
 permit host 192.168.100.100                  ! Allow this specific IP
 deny any                                     ! Block all other traffic
 exit                                         ! Return to global config
access-list 110 permit ip 192.168.1.0 0.0.0.255 192.168.3.0 0.0.0.255  ! Define VPN interesting traffic
crypto isakmp policy 10                       ! Create IKE Phase 1 policy
 encryption aes 256                           ! Set encryption method
 authentication pre-share                     ! Set authentication method
 group 5                                      ! Set Diffie-Hellman group
 exit                                         ! Return to global config
crypto isakmp key vpnpa55 address 10.2.2.2    ! Set the pre-shared key matching R3's IP
crypto ipsec transform-set VPN-SET esp-aes esp-sha-hmac  ! Define IKE Phase 2 parameters
crypto map VPN-MAP 10 ipsec-isakmp            ! Create the crypto map
 description VPN connection to R3             ! Add description
 set peer 10.2.2.2                            ! Specify R3 as the VPN peer
 set transform-set VPN-SET                    ! Apply the transform-set created above
 match address 110                            ! Link to the interesting traffic ACL
 exit                                         ! Return to global config
interface S0/0/0                              ! Enter the serial interface connecting to R3
 crypto map VPN-MAP                           ! Apply the crypto map to the interface
 exit                                         ! Return to global config


 7. ROUTER 3 SPECIFIC (R3): VPN Config
